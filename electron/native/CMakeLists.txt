cmake_minimum_required(VERSION 3.20)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "")
endif()
project(TextifyElectronNative LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_METAL ${APPLE} CACHE BOOL "" FORCE)
if(NOT APPLE)
  set(GGML_VULKAN ON CACHE BOOL "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.native/whisper/CMakeLists.txt")
  message(FATAL_ERROR "Run npm run native:prepare first. Build never downloads source.")
endif()
add_subdirectory(../.native/whisper whisper-build)
add_executable(textify-whisper whisper-worker.cpp)
target_link_libraries(textify-whisper PRIVATE whisper)
if(APPLE)
  set_source_files_properties(whisper-worker.cpp PROPERTIES LANGUAGE OBJCXX)
  target_compile_options(textify-whisper PRIVATE -fobjc-arc)
  target_link_libraries(textify-whisper PRIVATE "-framework Metal")
endif()
add_executable(textify-gpu-policy-test gpu-policy-test.cpp)
target_link_libraries(textify-gpu-policy-test PRIVATE ggml)
enable_testing()
add_test(NAME gpu-required COMMAND textify-gpu-policy-test)
if(WIN32)
  # Check the system driver loader before using Vulkan; a missing DLL must
  # produce a structured GPU error, not a Windows startup-error dialog.
  target_link_libraries(textify-whisper PRIVATE delayimp)
  target_link_options(textify-whisper PRIVATE "/DELAYLOAD:vulkan-1.dll")
  target_link_libraries(textify-gpu-policy-test PRIVATE delayimp)
  target_link_options(textify-gpu-policy-test PRIVATE "/DELAYLOAD:vulkan-1.dll")
endif()
if(APPLE)
  add_executable(textify-platform platform-mac.mm)
  target_compile_options(textify-platform PRIVATE -fobjc-arc)
  target_link_libraries(textify-platform PRIVATE "-framework AppKit" "-framework ApplicationServices")
elseif(WIN32)
  add_executable(textify-platform platform-win.cpp)
  target_link_libraries(textify-platform PRIVATE user32 ole32 oleaut32)
elseif(UNIX)
  find_package(X11 REQUIRED)
  add_executable(textify-platform platform-x11.cpp)
  target_link_libraries(textify-platform PRIVATE X11::X11)
endif()
